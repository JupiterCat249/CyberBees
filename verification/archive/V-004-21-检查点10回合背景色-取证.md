**用代码创建背景条**（不改场景文件 → 更小风险、更易回滚）：

```gdscript
var _turn_bg: ColorRect = null

func _ensure_turn_bg() -> void:
	if _turn_bg != null and is_instance_valid(_turn_bg): return
	if overlay == null: return
	_turn_bg = ColorRect.new()
	_turn_bg.name = "TurnBg"
	_turn_bg.position = Vector2(440.0, 0.0)      # 与 info 文本同位
	_turn_bg.size = Vector2(700.0, 54.0)
	_turn_bg.mouse_filter = Control.MOUSE_FILTER_IGNORE   # **绝不拦截点击**
	overlay.add_child(_turn_bg)
	overlay.move_child(_turn_bg, 0)              # 置于最底层，作为信息条背景
```
`_on_state_changed()` 中与文本色一同更新：
```gdscript
	var _over: bool = state.winner != ""
	var _tcol: Color = Color(1,1,1,0.5) if _over else state.anim.turn_color(state.current == "green")
	_labels["info"].add_theme_color_override("font_color", _tcol)     # 文本态
	_ensure_turn_bg()
	_turn_bg.color = Color(_tcol.r, _tcol.g, _tcol.b, 0.2 if _over else 0.5)   # 背景态（半透明，不压文字）
```
